cmake_minimum_required(VERSION 3.24)
project(linux_scripts_dock VERSION 0.5.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Gui Qml Quick Network)
find_program(CARGO_EXECUTABLE cargo REQUIRED)
find_program(MOLD_EXECUTABLE mold REQUIRED)
add_link_options(-fuse-ld=mold)
find_program(QMAKE_EXECUTABLE NAMES qmake6 qmake REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(GIO REQUIRED IMPORTED_TARGET gio-2.0)
set(RUST_TARGET "${CMAKE_CURRENT_BINARY_DIR}/cargo")
set(RUST_EXPORT "${CMAKE_CURRENT_BINARY_DIR}/cxxqt")
file(GLOB_RECURSE RUST_SOURCES CONFIGURE_DEPENDS "rust/src/*.rs")
add_custom_command(OUTPUT "${RUST_TARGET}/release/libdock_rust.a"
    COMMAND ${CMAKE_COMMAND} -E env
        "QMAKE=${QMAKE_EXECUTABLE}" "CARGO_TARGET_DIR=${RUST_TARGET}" "CARGO_BUILD_JOBS=2"
        "CXX_QT_EXPORT_DIR=${RUST_EXPORT}" "CXX_QT_EXPORT_CRATE_dock-rust=1"
        ${CARGO_EXECUTABLE} build --manifest-path "${CMAKE_CURRENT_SOURCE_DIR}/rust/Cargo.toml" --features qt --release --locked
    DEPENDS ${RUST_SOURCES} rust/Cargo.toml rust/Cargo.lock rust/build.rs
    WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}"
    VERBATIM USES_TERMINAL)
add_custom_target(dock-rust-build DEPENDS "${RUST_TARGET}/release/libdock_rust.a")
add_library(dock_rust STATIC IMPORTED)
set_target_properties(dock_rust PROPERTIES IMPORTED_LOCATION "${RUST_TARGET}/release/libdock_rust.a")
add_dependencies(dock_rust dock-rust-build)
set(ADAPTER_SOURCES src/runtime.cpp src/runtime.h src/applicationmodel.cpp src/applicationmodel.h src/platform/x11/windowmodel.cpp src/platform/x11/windowmodel.h)
qt_add_executable(linux-scripts-dock src/main.cpp src/launcheripc.cpp src/launcheripc.h ${ADAPTER_SOURCES})
add_dependencies(linux-scripts-dock dock-rust-build)
target_include_directories(linux-scripts-dock PRIVATE src "${RUST_EXPORT}/crates/dock-rust/include")
qt_add_resources(linux-scripts-dock "ui" PREFIX "/" FILES qml/Main.qml)
target_link_libraries(linux-scripts-dock PRIVATE dock_rust Qt6::Core Qt6::Gui Qt6::Qml Qt6::Quick Qt6::Network PkgConfig::GIO ${CMAKE_DL_LIBS} pthread)
target_compile_options(linux-scripts-dock PRIVATE -Wall -Wextra -Wpedantic)
install(TARGETS linux-scripts-dock RUNTIME DESTINATION bin)
include(CTest)
if(BUILD_TESTING)
    find_package(Qt6 6.8 REQUIRED COMPONENTS Test)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    add_test(NAME dock-thunar-action COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/thunar-action-test.py)
    qt_add_executable(dock-launcher-ipc-test tests/launcheripc-test.cpp src/launcheripc.cpp src/launcheripc.h)
    target_include_directories(dock-launcher-ipc-test PRIVATE src)
    target_link_libraries(dock-launcher-ipc-test PRIVATE Qt6::Core Qt6::Network)
    add_test(NAME dock-launcher-ipc COMMAND dock-launcher-ipc-test)
    add_test(NAME dock-rust-core COMMAND ${CARGO_EXECUTABLE} test --manifest-path "${CMAKE_CURRENT_SOURCE_DIR}/rust/Cargo.toml" --locked)
    qt_add_executable(dock-qml-smoke tests/qml-smoke.cpp)
    qt_add_resources(dock-qml-smoke "test_ui" PREFIX "/" FILES qml/Main.qml)
    target_link_libraries(dock-qml-smoke PRIVATE Qt6::Core Qt6::Gui Qt6::Qml Qt6::Quick Qt6::Test)
    add_test(NAME dock-qml-smoke COMMAND dock-qml-smoke)
    set_tests_properties(dock-qml-smoke PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software")
    pkg_check_modules(XCB REQUIRED IMPORTED_TARGET xcb)
    qt_add_executable(dock-x11-grab-test tests/x11-grab-test.cpp ${ADAPTER_SOURCES})
    add_dependencies(dock-x11-grab-test dock-rust-build)
    target_include_directories(dock-x11-grab-test PRIVATE src "${RUST_EXPORT}/crates/dock-rust/include")
    target_link_libraries(dock-x11-grab-test PRIVATE dock_rust Qt6::Core Qt6::Gui Qt6::Qml Qt6::Quick Qt6::Network PkgConfig::GIO PkgConfig::XCB ${CMAKE_DL_LIBS} pthread)
endif()
